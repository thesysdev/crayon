

root = Card([header, thresholdAlert, searchForm, lowStockTable, reorderNote, followUps], "card", "column", "l")
header = CardHeader("Low Inventory Warning", "Items that need replenishment")
thresholdAlert = Alert("Low-stock threshold", "Warning is triggered when an item has fewer than 20 units remaining.", "warning")
searchForm = Form("inventory-search", searchFormButtons, [searchQueryField])
searchFormButtons = Buttons([searchButton, resetButton])
searchButton = Button("Search", null, "primary")
resetButton = Button("Reset", null, "secondary")
searchQueryField = FormControl("Search items", searchQueryInput, "Search by item name, SKU, or supplier")
searchQueryInput = Input("query", "e.g. USB-C cable", "text")
lowStockTable = Table([itemColumn, remainingColumn, thresholdColumn, reorderColumn])
itemColumn = Col("Item", ["Wireless Mouse", "Mechanical Keyboard", "USB-C Cable", "27-inch Monitor"])
remainingColumn = Col("Remaining units", [8, 12, 5, 17], "number")
thresholdColumn = Col("Warning threshold", [20, 20, 20, 20], "number")
reorderColumn = Col("Reorder", ["Reorder Wireless Mouse", "Reorder Mechanical Keyboard", "Reorder USB-C Cable", "Reorder 27-inch Monitor"], "action")
reorderNote = TextContent("Select a reorder action to create a purchase order for the selected item.", "small")
followUps = FollowUpBlock([showAllInventory, changeThreshold])
showAllInventory = FollowUpItem("Show all inventory")
changeThreshold = FollowUpItem("Change low-stock threshold")
