  <script>
    const byId = (id) => document.getElementById(id);

    const applicationForm = byId("applicationForm");
    const fullName = byId("fullName");
    const email = byId("email");
    const resumeLink = byId("resumeLink");

    const applyStatus = byId("applyStatus");
    const shareStatus = byId("shareStatus");

    const copyLinkButton = byId("copyLinkButton");
    const emailShareButton = byId("emailShareButton");
    const linkedinShareLink = byId("linkedinShareLink");
    const xShareLink = byId("xShareLink");
    const nativeShareButton = byId("nativeShareButton");

    const shareUrl = window.location.href.split("#")[0];
    const shareTitle = document.title;
    const shareBody = `Check out this job posting: ${shareTitle}\n\n${shareUrl}`;

    linkedinShareLink.href =
      `https://www.linkedin.com/sharing/share-offsite/?url=${encodeURIComponent(shareUrl)}`;

    xShareLink.href =
      `https://twitter.com/intent/tweet?text=${encodeURIComponent(shareTitle)}&url=${encodeURIComponent(shareUrl)}`;

    function setStatus(element, message, isError = false) {
      element.textContent = message;
      element.className = `status ${isError ? "error" : "success"}`;

      if (element.timeoutId) clearTimeout(element.timeoutId);
      element.timeoutId = setTimeout(() => {
        if (element.textContent === message) element.textContent = "";
      }, 5000);
    }

    async function copyPostingLink() {
      try {
        if (navigator.clipboard?.writeText) {
          await navigator.clipboard.writeText(shareUrl);
        } else {
          const temporaryInput = document.createElement("textarea");
          temporaryInput.value = shareUrl;
          temporaryInput.setAttribute("readonly", "");
          temporaryInput.style.position = "absolute";
          temporaryInput.style.left = "-9999px";

          document.body.appendChild(temporaryInput);
          temporaryInput.select();
          document.execCommand("copy");
          temporaryInput.remove();
        }

        setStatus(shareStatus, "Posting link copied to clipboard.");
      } catch {
        setStatus(shareStatus, `Copy this link: ${shareUrl}`, true);
      }
    }

    copyLinkButton.addEventListener("click", copyPostingLink);

    emailShareButton.addEventListener("click", () => {
      window.location.href =
        `mailto:?subject=${encodeURIComponent(shareTitle)}&body=${encodeURIComponent(shareBody)}`;
    });

    nativeShareButton.addEventListener("click", async () => {
      if (navigator.share) {
        try {
          await navigator.share({
            title: shareTitle,
            text: shareBody,
            url: shareUrl
          });
        } catch {
          // User cancelled the share prompt.
        }
      } else {
        await copyPostingLink();
      }
    });

    applicationForm.addEventListener("submit", (event) => {
      event.preventDefault();

      if (!applicationForm.checkValidity()) {
        applicationForm.reportValidity();
        setStatus(applyStatus, "Please complete all required fields.", true);
        return;
      }

      const payload = {
        role: shareTitle,
        name: fullName.value.trim(),
        email: email.value.trim(),
        resume: resumeLink.value.trim(),
        submittedAt: new Date().toISOString()
      };

      // Front-end only demo.
      // Connect this payload to your API, ATS, or form backend here.
      console.log("Application payload:", payload);

      setStatus(
        applyStatus,
        `Thanks ${payload.name}. Your application was captured. Resume link: ${payload.resume}`
      );

      applicationForm.reset();
    });
  </script>

All six requested items are included:

1. **Role title and team** — hero header with title and team.
2. **Short role description** — “Role description” panel.
3. **Main qualifications sought** — qualifications list.
4. **Salary range** — compensation panel.
5. **Apply with name, email, resume link** — application form.
6. **Share posting** — copy link, email, LinkedIn, X/Twitter, and native share fallback.
